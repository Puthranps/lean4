cmake_minimum_required(VERSION 3.20)
project(lean_windows_sandbox_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The native prototype requires Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(lean-sandbox-prototype STATIC sandbox.cpp)
target_compile_features(lean-sandbox-prototype PRIVATE cxx_std_17)
target_compile_definitions(lean-sandbox-prototype PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(lean-sandbox-prototype PRIVATE advapi32 userenv ole32)
if(MSVC)
  target_compile_options(lean-sandbox-prototype PRIVATE /W4 /WX /EHsc)
endif()
add_executable(sandbox-driver driver.cpp)
target_compile_features(sandbox-driver PRIVATE cxx_std_17)
target_compile_definitions(sandbox-driver PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(sandbox-driver PRIVATE lean-sandbox-prototype windowsapp)
add_executable(probe ../../tests/check-sandbox-confinement/probe.cpp)
target_compile_features(probe PRIVATE cxx_std_17)
target_link_libraries(probe PRIVATE ws2_32)
if(MSVC)
  target_compile_options(sandbox-driver PRIVATE /W4 /WX /EHsc)
  target_compile_options(probe PRIVATE /W4 /WX /EHsc)
endif()
